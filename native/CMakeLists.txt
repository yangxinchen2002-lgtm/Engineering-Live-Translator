cmake_minimum_required(VERSION 3.24)
project(elt_native LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "ELT requires Windows and MSVC with Windows SDK >= 10.0.20348")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(elt_capture audio_capture/main.cpp)
target_compile_definitions(elt_capture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(elt_capture PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(elt_capture PRIVATE mmdevapi ole32 advapi32 uuid)
enable_testing()
add_executable(protocol_test audio_capture/protocol_test.cpp)
target_compile_options(protocol_test PRIVATE /W4 /EHsc)
add_test(NAME native_protocol COMMAND protocol_test)
add_executable(queue_test audio_capture/queue_test.cpp)
target_compile_options(queue_test PRIVATE /W4 /EHsc)
add_test(NAME native_queue COMMAND queue_test)
add_executable(source_policy_test audio_capture/source_policy_test.cpp)
target_compile_options(source_policy_test PRIVATE /W4 /EHsc)
add_test(NAME native_source_policy COMMAND source_policy_test)
