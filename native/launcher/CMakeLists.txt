cmake_minimum_required(VERSION 3.24)
project(elt_launcher LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "ELT launcher requires Windows x64 / MSVC")
endif()
add_executable(ELT WIN32 main.cpp app.rc)
target_compile_features(ELT PRIVATE cxx_std_20)
set_property(TARGET ELT PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(ELT PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ELT PRIVATE /W4 /permissive- /EHsc /utf-8)
target_include_directories(ELT PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ELT PRIVATE shell32 ole32 user32 uuid)
